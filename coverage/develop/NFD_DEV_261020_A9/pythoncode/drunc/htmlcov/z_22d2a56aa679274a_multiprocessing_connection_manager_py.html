38 def create_process( 

39 self, 

40 process_id: str, 

41 target_func: TargetFunc[P, object], 

42 *args: object, 

43 **kwargs: object, 

44 ) -> RunningGrpcServer: 

45 """ 

46 Create a multiprocessing.Process handle with coordination events. 

47 

48 Creates ready and stop events for process lifecycle coordination, 

49 wraps the target function to set environment variables, and prepares 

50 the process for execution. Events are appended to the argument list 

51 so the server functions can use them for signalling. 

52 

53 Args: 

54 process_id: Unique identifier for the process 

55 target_func: Function to execute in the new process 

56 *args: Arguments to pass to target function 

57 **kwargs: Keyword arguments to pass to target function 

58 

59 Returns: 

60 RunningGrpcServer containing the multiprocessing.Process and events 

61 """ 

62 # Create coordination events for this process 

63 ready_event: multiprocessing.synchronize.Event = multiprocessing.Event() 

64 stop_event: multiprocessing.synchronize.Event = multiprocessing.Event() 

65 

66 # Wrap target function to set environment variables 

67 def wrapped_target(*target_args: P.args, **target_kwargs: P.kwargs) -> object: 

68 for key, value in self.env_vars.items(): 

69 os.environ[key] = value 

70 return target_func(*target_args, **target_kwargs) 

71 

72 # Append events to arguments so server functions can use them 

73 extended_args = args + (ready_event, stop_event) 

74 

75 # Create process handle with wrapped function and extended arguments 

76 handle = RunningGrpcServer(process_id, wrapped_target, extended_args, kwargs) 

77 

78 # Create multiprocessing.Process 

79 mp_process = multiprocessing.Process( 

80 target=wrapped_target, args=extended_args, kwargs=kwargs, name=process_id 

81 ) 

82 

83 handle.set_process(mp_process) 

84 

85 # Store events in handle for access by server manager 

86 handle.ready_event = ready_event 

87 handle.stop_event = stop_event 

88 

89 self.process_handles[process_id] = handle 

90 

91 return handle

93 def start_process(self, handle: RunningGrpcServer) -> None: 

94 """ 

95 Start a multiprocessing.Process. 

96 

97 Args: 

98 handle: RunningGrpcServer containing multiprocessing.Process 

99 

100 Raises: 

101 RuntimeError: If process is already started or cannot start 

102 """ 

103 if handle.started: 

104 raise RuntimeError(f"Process {handle.process_id} is already started") 

105 

106 try: 

107 process = handle.process 

108 if process is None: 

109 raise RuntimeError(f"Process {handle.process_id} has not been created") 

110 process.start() 

111 handle.mark_started() 

112 except Exception as e: 

113 raise RuntimeError(f"Failed to start process {handle.process_id}: {e}")

115 def stop_process(self, handle: RunningGrpcServer, timeout: float = 10.0) -> None: 

116 """ 

117 Stop a multiprocessing.Process gracefully. 

118 

119 Signals the stop event if present to allow graceful shutdown, then 

120 attempts termination. Forces kill if process doesn't terminate within 

121 the timeout period. 

122 

123 Args: 

124 handle: RunningGrpcServer containing the process to stop 

125 timeout: Maximum time to wait for graceful shutdown 

126 

127 Raises: 

128 RuntimeError: If process cannot be stopped 

129 """ 

130 if not handle.started or not handle.process: 

131 return 

132 

133 process = handle.process 

134 

135 if not process.is_alive(): 

136 return 

137 

138 # Signal graceful shutdown if event exists 

139 if handle.stop_event: 

140 handle.stop_event.set() 

141 

142 # Attempt graceful termination 

143 process.terminate() 

144 process.join(timeout=timeout) 

145 

146 # Force kill if still alive 

147 if process.is_alive(): 

148 process.kill() 

149 process.join(timeout=2.0)
