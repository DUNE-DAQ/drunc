113def load_ssh_config() -> paramiko.SSHConfig: 

114 """ 

115 Load the user's SSH configuration from ~/.ssh/config using Paramiko's SSHConfig 

116 class. 

117 

118 This function reads the SSH configuration file and parses it to create an SSHConfig 

119 object that can be used to look up host-specific settings when connecting to hosts. 

120 

121 Args: 

122 None 

123 

124 Returns: 

125 paramiko.SSHConfig: An SSHConfig object containing the parsed SSH configuration. 

126 

127 Raises: 

128 FileNotFoundError: If the SSH configuration file is not found at the expected 

129 location. 

130 """ 

131 # Determine the absolute path to the SSH configuration file. 

132 config_path = os.path.expanduser("~/.ssh/config") 

133 if not os.path.exists(config_path): 

134 raise FileNotFoundError(f"SSH config file not found at {config_path}") 

135 

136 # Create an SSHConfig object and parse the SSH configuration file to populate it 

137 # with the host-specific settings. 

138 ssh_config = paramiko.SSHConfig() 

139 

140 # Open the SSH configuration file and parse it to populate the SSHConfig object. 

141 try: 

142 ssh_config.parse(open(config_path)) 

143 except Exception as e: 

144 raise Exception(f"Error parsing SSH config file: {e}") 

145 

146 return ssh_config

149def ping_host(hostname: str) -> bool: 

150 """ 

151 Ping a host to check if it is reachable. 

152 

153 This function uses the system's ping command to send a single ICMP echo request to 

154 the specified hostname. Sends a single packet and waits for a response for up to 1 

155 second. 

156 

157 Args: 

158 hostname: The hostname or IP address of the host to ping. 

159 

160 Returns: 

161 bool: True if the host is reachable (pingable), False otherwise. 

162 

163 Raises: 

164 subprocess.CalledProcessError: If the ping command fails to execute properly. 

165 """ 

166 command = ["ping", "-c", "1", "-W", "1", hostname] 

167 return ( 

168 subprocess.call(command, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL) 

169 == 0 

170 )

173def get_host_metrics(host_alias: str, ssh_config: paramiko.SSHConfig) -> HostData: 

174 """ 

175 For a given host alias, perform the following checks: 

176 - Ping the host to determine if it's online. 

177 - If pingable, attempt an SSH connection using Paramiko. 

178 - If SSH is successful, gather CPU, RAM, uptime, and NFS status information. 

179 

180 Args: 

181 host_alias: The SSH alias of the host to check. 

182 ssh_config: The loaded SSH configuration for connection details. 

183 

184 Returns: 

185 A dictionary with the results of the checks and gathered metrics for the host. 

186 

187 Raises: 

188 Any exceptions during SSH connection or command execution are caught and used 

189 in the returned dictionary under the 'cpu_model' key for visibility in the UI. 

190 """ 

191 

192 # Start with a copy of the default host data and set the alias 

193 result = DEFAULT_HOST_DATA.copy() 

194 result["alias"] = host_alias 

195 

196 # Look up the host configuration from the SSH config using the provided host alias. 

197 host_conf = ssh_config.lookup(host_alias) 

198 

199 def _normalise_ssh_conf_str(value: object, default: str) -> str: 

200 """ 

201 Normalise Paramiko SSH config values to a single string. 

202 

203 Args: 

204 value: The value from the SSH config, which may be a string or a list. 

205 default: The default string to return if the value is not a string or list. 

206 

207 Returns: 

208 str: The normalised string value from the SSH config, or the default if not 

209 applicable. 

210 

211 Raises: 

212 None. This function handles type checking and returns a default value if the 

213 input is not a string or a non-empty list. 

214 """ 

215 if isinstance(value, str): 

216 return value 

217 if isinstance(value, list) and value: 

218 first = value[0] 

219 if isinstance(first, str): 

220 return first 

221 return default 

222 

223 # Determine the real hostname to connect to. If the SSH config provides a "hostname" 

224 # entry for this alias, use that; otherwise, use the alias itself as the hostname. 

225 # Note - this casting is required because Paramiko's SSHConfig.lookup() can return 

226 # a list for some fields, and we want to ensure we have a single string for the 

227 # hostname. 

228 hostname = _normalise_ssh_conf_str(host_conf.get("hostname"), host_alias) 

229 

230 # First, check if the host is pingable before attempting an SSH connection. If the 

231 # host is not pingable, attempting the SSH connection can be skipped and it can be 

232 # marked as offline. 

233 result["is_pingable"] = ping_host(hostname) 

234 if not result["is_pingable"]: 

235 result["details"] = "No ICMP Response" 

236 return result 

237 

238 # Initialize the SSH client 

239 client = paramiko.SSHClient() 

240 

241 # Set the custom missing host key policy to track and update the result dictionary 

242 client.set_missing_host_key_policy(TrackingAutoAddPolicy(result)) 

243 

244 # Look up the host configuration from the SSH config using the provided alias. This 

245 # will allow us to retrieve the real hostname, username, port, and key file to use 

246 # for the connection. If the alias is not found in the SSH config, we will use the 

247 # alias itself as the hostname. 

248 try: 

249 # Load system host keys to ensure we have the latest known hosts information. If 

250 # this fails, use the default behavior of the SSH client, which will handle missing 

251 # keys according to the policy set below. 

252 client.load_system_host_keys() 

253 

254 username = _normalise_ssh_conf_str(host_conf.get("user"), os.getlogin()) 

255 port_raw = host_conf.get("port") 

256 port = int(port_raw) if isinstance(port_raw, (str, int)) else 22 

257 key_filename = ( 

258 _normalise_ssh_conf_str(host_conf.get("identityfile"), "") or None 

259 ) 

260 

261 # Attempt to establish an SSH connection to the host using the prepared 

262 # arguments. If the host key is missing, the custom policy will handle it and 

263 # update the result dict. 

264 client.connect( 

265 hostname=hostname, 

266 username=username, 

267 port=port, 

268 timeout=5.0, 

269 key_filename=key_filename, 

270 ) 

271 

272 # If the connection was successfully established and the key was valid, the 

273 # connection is verified and can be marked as such. 

274 if result["ssh_key"] == "Pending": 

275 result["ssh_key"], result["key_color"] = "Verified", "green" 

276 

277 # Construct the command to gather the relevant metrics 

278 cmd = ( 

279 r"lscpu | grep -P '^Socket\(s\):|Vendor ID:|Model name:|Core\(s\) per socket:'; " 

280 "uptime -p; " 

281 "top -bn1 | grep 'Cpu(s)' | awk '{print $2 + $4}'; " 

282 "free -g | awk '/Mem:/ {print $2,$3}'; " 

283 "[ -d /nfs ] && [ \"$(ls -A /nfs)\" ] && echo 'NFS_OK' || echo 'NFS_MISSING'" 

284 ) 

285 

286 # Execute the command and read the output 

287 _, stdout, _ = client.exec_command(cmd) 

288 lines = stdout.read().decode().strip().splitlines() 

289 

290 # Define temporary variables to hold CPU socket and core information for 

291 # calculating total cores 

292 sockets: int = 0 

293 cores_per_socket: int = 0 

294 

295 # Parse the output lines to extract the relevant metrics and update the result 

296 # dictionary 

297 for line in lines: 

298 # 1. Physical CPU Sockets 

299 if "Socket(s):" in line: 

300 try: 

301 sockets = int(line.split(":")[1].strip()) 

302 result["cpu_count"] = str(sockets) 

303 except ValueError: 

304 pass 

305 

306 elif "Core(s) per socket:" in line: 

307 try: 

308 cores_per_socket = int(line.split(":")[1].strip()) 

309 except ValueError: 

310 pass 

311 

312 # 2. CPU Vendor/Color 

313 elif "Vendor ID:" in line: 

314 result["model_color"] = "bold red" if "AMD" in line else "bold blue" 

315 

316 # 3. CPU Model Name 

317 elif "Model name:" in line: 

318 result["cpu_model"] = line.split(":")[1].strip() 

319 

320 # 4. Uptime (looks for the "up" prefix from uptime -p) 

321 elif line.startswith("up "): 

322 result["uptime"] = line.replace("up ", "") 

323 

324 # 5. NFS Status 

325 elif "NFS_" in line: 

326 result["nfs_ok"] = line == "NFS_OK" 

327 

328 # 6. RAM (Looking for the line with two integers: Total Used) 

329 elif " " in line and "." not in line and not line.startswith("up"): 

330 parts = line.split() 

331 if len(parts) == 2: 

332 try: 

333 total, used = int(parts[0]), int(parts[1]) 

334 result["ram_total"] = f"{total}GB" 

335 result["ram_usage"] = ( 

336 f"{(used / total) * 100:.1f}%" if total > 0 else "0%" 

337 ) 

338 except ValueError: 

339 pass 

340 

341 # 7. CPU Usage (The only line left that should be a float/decimal) 

342 else: 

343 try: 

344 # Check if the line is purely a number (CPU percentage) 

345 float_val = float(line) 

346 result["cpu_usage"] = f"{float_val}%" 

347 except ValueError: 

348 pass 

349 

350 if sockets > 0 and cores_per_socket > 0: 

351 result["total_cores"] = str(sockets * cores_per_socket) 

352 

353 # Handle the case where the host key does not match the expected key in the known 

354 # hosts file. This indicates a potential security issue, and we will update the 

355 # result dictionary to reflect that the key is a mismatch and the host is down. 

356 except paramiko.BadHostKeyException: 

357 result["ssh_key_status"] = "MISMATCH" 

358 result["ssh_key_color"] = "bold red" 

359 result["details"] = "Host Key Changed!" 

360 

361 # Handle authentication failures, which indicate that the host is offline or the key 

362 # is not valid for this host. 

363 except paramiko.AuthenticationException: 

364 result["ssh_key_status"] = "Verified" 

365 result["ssh_key_color"] = "green" 

366 result["details"] = "Auth Failed" 

367 

368 # Handle SSH exceptions, which can occur for various reasons such as network issues, 

369 # SSH service not running on the host, or other SSH-related problems. 

370 except paramiko.SSHException as e: 

371 result["status"] = "OFFLINE" 

372 result["details"] = f"SSH Error: {str(e)[:20]}" 

373 

374 # Handle any other exceptions that occur during the connection attempt. Treat this 

375 # as an indication that the host is offline or unreachable. 

376 except Exception: 

377 result["details"] = "Conn Error" 

378 finally: 

379 client.close() 

380 

381 return result

384def generate_table(results_map: dict[str, HostData]) -> Table: 

385 """ 

386 Generate a Rich Table object to display the status of the NP0x cluster hosts. 

387 

388 This function creates a Rich Table with columns for Host, Status, User SSH Key 

389 Status, CPU Model/Details, and Uptime. It iterates through results_map, and updates 

390 the table rows based on the its content. 

391 

392 Args: 

393 results_map (dict[str, str]): A dictionary mapping host aliases to their status 

394 information, including connection status, key verification status, CPU 

395 details, and uptime. 

396 

397 Returns: 

398 Table: A Rich Table object populated with the status information for each host 

399 in the NP0x cluster, ready to be rendered in the console. 

400 """ 

401 

402 # QOL feature 

403 n_online = sum(1 for r in results_map.values() if r["is_pingable"] is True) 

404 n_hosts: int = len(results_map) 

405 

406 # Create a Rich Table with appropriate columns and styling to display the host 

407 # status information. 

408 table = Table( 

409 title=f"[bold cyan]ProtoDUNE Cluster ({n_online}/{n_hosts} Online)[/]", 

410 box=box.ROUNDED, 

411 header_style="bold cyan", 

412 ) 

413 

414 # Define the columns for the table. 

415 table.add_column("Host", justify="center", style="magenta") 

416 table.add_column("Ping Status", justify="center") 

417 table.add_column("SSH Key", justify="center") 

418 table.add_column("CPUs", justify="center") 

419 table.add_column("Hardware / Error Details", justify="center") 

420 table.add_column("Total cores", justify="center") 

421 table.add_column("CPU %", justify="center") 

422 table.add_column("RAM", justify="center") 

423 table.add_column("RAM %", justify="center") 

424 table.add_column("NFS mounted", justify="center") 

425 table.add_column("Uptime", justify="center") 

426 

427 # Iterate through the results_map and add a row to the table for each host. 

428 for host in NP0X_CLUSTER_HOSTS: 

429 # Retrieve the result dictionary for the current host 

430 r = results_map.get(host, DEFAULT_HOST_DATA) 

431 hostname = r["alias"] if r["alias"] else host 

432 

433 # Host is Pingable 

434 if r["is_pingable"] is True: 

435 ping_cell = MARK_CHECK 

436 key_cell = f"[{r['key_color']}]{r['ssh_key']}[/]" 

437 nfs_cell = MARK_CHECK if r["nfs_ok"] else MARK_CROSS 

438 details = f"[{r['model_color']}]{r['cpu_model']}[/]" 

439 uptime, c_ld, r_ld = ( 

440 f"[dim]{r['uptime']}[/]", 

441 r["cpu_usage"], 

442 r["ram_usage"], 

443 ) 

444 

445 # Scan in Progress 

446 elif r["is_pingable"] is None: 

447 ping_cell = "[dim white]WAITING[/]" 

448 key_cell = nfs_cell = details = uptime = c_ld = r_ld = "-" 

449 details = "[dim italic]Scanning...[/]" 

450 

451 # Host is Unreachable (Offline) 

452 else: 

453 ping_cell = MARK_CROSS 

454 key_cell = nfs_cell = uptime = c_ld = r_ld = "-" 

455 details = f"[bold yellow]{r['cpu_model']}[/]" 

456 

457 table.add_row( 

458 hostname, 

459 ping_cell, 

460 key_cell, 

461 r["cpu_count"], 

462 details, 

463 r["total_cores"], 

464 c_ld, 

465 r["ram_total"], 

466 r_ld, 

467 nfs_cell, 

468 uptime, 

469 ) 

470 return table

473def main() -> None: 

474 """ 

475 Main function to execute the NP0x cluster status check and display results in a 

476 live-updating table. 

477 

478 This function initializes the console and SSH configuration, sets up a results map 

479 to track the status of each host, and uses a ThreadPoolExecutor to concurrently 

480 check the status of each host in the NP0x cluster. The results are displayed in a 

481 live-updating table using Rich's Live feature, which refreshes the display as 

482 results come in. 

483 

484 Args: 

485 None 

486 

487 Returns: 

488 None 

489 

490 Raises: 

491 Any exceptions that occur during the execution of the host checks will be 

492 handled within the get_host_info function. 

493 """ 

494 

495 # Initialize the console for Rich output and load the SSH configuration. 

496 console = Console() 

497 console.print("") # Buffer line for better aesthetics 

498 ssh_config = load_ssh_config() 

499 

500 # Define the default results map with initial values for each host. This map will be 

501 # updated as results come in from the concurrent checks. 

502 results_map = { 

503 h: DEFAULT_HOST_DATA.copy() | {"alias": h} for h in NP0X_CLUSTER_HOSTS 

504 } 

505 

506 # Use Rich's Live to create a live-updating table. The table will be refreshed as 

507 # results come in from the concurrent checks. 

508 with Live( 

509 generate_table(results_map), console=console, refresh_per_second=10 

510 ) as live: 

511 with ThreadPoolExecutor(max_workers=15) as executor: 

512 # Map each host to a future that will execute the get_host_info function 

513 # concurrently. 

514 futures = { 

515 executor.submit(get_host_metrics, h, ssh_config): h 

516 for h in NP0X_CLUSTER_HOSTS 

517 } 

518 

519 # As each future completes, update the results map with the new information 

520 # and refresh the live table to reflect the updated status of the hosts. 

521 for f in as_completed(futures): 

522 results_map[futures[f]] = f.result() 

523 live.update(generate_table(results_map)) 

524 

525 console.print("\n[bold green]Scan Complete.[/]")

529 main()
